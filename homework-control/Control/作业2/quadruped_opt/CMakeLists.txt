cmake_minimum_required(VERSION 3.16)
project(quadruped_opt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
set(OSQP_BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(OSQP_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/osqp EXCLUDE_FROM_ALL)
add_executable(quadruped_opt main.cpp)
target_include_directories(quadruped_opt PRIVATE third_party/osqp/include)
target_link_libraries(quadruped_opt PRIVATE Eigen3::Eigen osqpstatic)
